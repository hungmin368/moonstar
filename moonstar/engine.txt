/* ---------- 配色 ---------- */
function hexHue(hex){
  const r=parseInt(hex.slice(1,3),16)/255, g=parseInt(hex.slice(3,5),16)/255, b=parseInt(hex.slice(5,7),16)/255;
  const mx=Math.max(r,g,b), mn=Math.min(r,g,b); if(mx===mn) return 0;
  const d=mx-mn;
  let h;
  if(mx===r) h=((g-b)/d)%6; else if(mx===g) h=(b-r)/d+2; else h=(r-g)/d+4;
  h*=60; if(h<0)h+=360; return h;
}
function hueDist(a,b){ const d=Math.abs(a-b)%360; return Math.min(d,360-d); }
function relL(hex){
  const r=parseInt(hex.slice(1,3),16)/255, g=parseInt(hex.slice(3,5),16)/255, b=parseInt(hex.slice(5,7),16)/255;
  return (Math.max(r,g,b)+Math.min(r,g,b))/2*100;
}
function hslToHex(h,s,l){
  s/=100; l/=100;
  const k=n=>(n+h/30)%12, a=s*Math.min(l,1-l);
  const f=n=>l-a*Math.max(-1,Math.min(k(n)-3,Math.min(9-k(n),1)));
  const to=x=>Math.round(x*255).toString(16).padStart(2,'0');
  return '#'+to(f(0))+to(f(8))+to(f(4));
}
function buildPalette(N){ /* 10×10 起動態均分色相 + 交替明度 */
  if(N<=9) return PALETTE;
  const M=N+4, out=[];
  for(let i=0;i<M;i++) out.push(hslToHex(i*360/M, 62, i%2?79:66));
  return out;
}
function assignRegionColors(N, regionOf, PAL){ /* 鄰接區塊自動拉開色相（10+ 併用明度差） */
  const adj=Array.from({length:N},()=>new Set());
  for(let i=0;i<N*N;i++){
    const r=Math.floor(i/N), c=i%N, rg=regionOf[i];
    if(c<N-1){ const o=regionOf[i+1]; if(o!==rg){ adj[rg].add(o); adj[o].add(rg); } }
    if(r<N-1){ const o=regionOf[i+N]; if(o!==rg){ adj[rg].add(o); adj[o].add(rg); } }
  }
  PAL = PAL || PALETTE;
  const info=PAL.map(hex=>({hex, hue:hexHue(hex), L:relL(hex)}));
  const useL = N>=10;
  const colorOf=new Array(N).fill(-1);
  const order=[...Array(N).keys()].sort((a,b)=>adj[b].size-adj[a].size);
  for(const rg of order){
    let best=-1, bestScore=-1;
    for(let p=0;p<PAL.length;p++){
      if(colorOf.includes(p)) continue;
      let score=999;
      for(const nb of adj[rg]){
        if(colorOf[nb]>=0){
          const dL=Math.abs(info[p].L-info[colorOf[nb]].L);
          const s=hueDist(info[p].hue,info[colorOf[nb]].hue)+((useL&&dL>=8)?22:0);
          score=Math.min(score,s);
        }
      }
      if(score>bestScore){ bestScore=score; best=p; }
    }
    colorOf[rg]=best;
  }
  return colorOf.map(p=>info[p].hex);
}

  o.start(t0); o.stop(t0 + dur + 0.05);
}

/* ---------- 盤面生成 ---------- */
function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; }
function neighbors4(i,N){ const r=Math.floor(i/N), c=i%N, out=[]; if(r>0)out.push(i-N); if(r<N-1)out.push(i+N); if(c>0)out.push(i-1); if(c<N-1)out.push(i+1); return out; }

function randomRegions(N){ /* 隨機連通區塊分割，每區至少2格 */
  const total=N*N;
  for(let t=0;t<200;t++){
    const regionOf=new Array(total).fill(-1);
    const seeds=shuffle([...Array(total).keys()]).slice(0,N);
    const frontiers=seeds.map(()=>[]);
    seeds.forEach((s,i)=>{ regionOf[s]=i; neighbors4(s,N).forEach(nb=>frontiers[i].push(nb)); });
    let assigned=N;
    while(assigned<total){
      let progressed=false;
      for(const r of shuffle([...Array(N).keys()])){
        if(assigned>=total) break;
        let cell=null;
        while(frontiers[r].length){
          const pick=Math.floor(Math.random()*frontiers[r].length);
          const c=frontiers[r][pick];
          frontiers[r].splice(pick,1);
          if(regionOf[c]!==-1) continue;
          cell=c; break;
        }
        if(cell===null) continue;
        regionOf[cell]=r; assigned++; progressed=true;
        neighbors4(cell,N).forEach(nb=>{ if(regionOf[nb]===-1) frontiers[r].push(nb); });
      }
      if(!progressed) break;
    }
    if(assigned<total) continue;
    const sizes=new Array(N).fill(0);
    regionOf.forEach(r=>sizes[r]++);
    if(sizes.some(s=>s<2)) continue;
    return regionOf;
  }
  return null;
}

function solve2(N, regionOf, nodeLimit, K){ /* v2.0.3：位元遮罩＋節點熔斷＋K 解上限 */
  const sols=[];
  const cur=new Uint8Array(N);
  let usedColMask=0, usedRegMask=0, nodes=0, aborted=false;
  const LIMIT = nodeLimit||Infinity;
  const MAXK = K||2;
  (function bt(r, prevCol){
    if(++nodes>LIMIT){ aborted=true; return; }
    if(sols.length>=MAXK) return;
    if(r===N){ sols.push(Array.from(cur)); return; }
    for(let c=0;c<N;c++){
      if(usedColMask>>c&1) continue;
      if(prevCol>=0 && Math.abs(c-prevCol)<=1) continue;
      const reg=regionOf[r*N+c];
      if(usedRegMask>>reg&1) continue;
      usedColMask|=1<<c; usedRegMask|=1<<reg; cur[r]=c;
      bt(r+1,c);
      usedColMask&=~(1<<c); usedRegMask&=~(1<<reg);
      if(sols.length>=2||aborted) return;
    }
  })(0,-1);
  return aborted ? null : sols;
}

function connectedAfterRemoval(N, regionOf, region, cellIdx){
  const cells=[];
  for(let i=0;i<N*N;i++) if(regionOf[i]===region && i!==cellIdx) cells.push(i);
  if(cells.length<2) return false;
  const seen=new Set([cells[0]]), st=[cells[0]];
  while(st.length){ const cur=st.pop(); for(const nb of neighbors4(cur,N)){ if(regionOf[nb]===region && nb!==cellIdx && !seen.has(nb)){ seen.add(nb); st.push(nb); } } }
  return seen.size===cells.length;
}

function generatePuzzle(N, deadline){ /* v1.1：定向獵殺 + 0解回退 + 硬限時 */
  deadline = deadline || 1200;
  const t0=Date.now();
  for(let restart=0;restart<400;restart++){
    if(Date.now()-t0>deadline) return null;
    const regionOf=randomRegions(N);
    if(!regionOf) continue;
    const nl = N>=14 ? 50000 : Infinity; /* 大盤節點熔斷 */
    let sols=solve2(N,regionOf,nl);
    if(sols===null){ continue; } /* 太慢：棄盤重來 */
    if(sols.length===0) continue;
    let zeroStreak=0;
    for(let iter=0;iter<800;iter++){
      if(iter%10===0 && Date.now()-t0>deadline) return null;
      if(sols.length===1 || (N>=16 && sols.length>=1)){
        /* N>=16：少解盤（1-4 解），貓取主解，全部解帶回 */
        let allSols=null;
        if(N>=16 && sols.length>1){
          const full=solve2(N, regionOf, 500000, 5);
          if(full && full.length>=1 && full.length<=4) allSols=full;
          else if(full && full.length===1){ sols=[full[0]]; }
          else { continue; }
        }
        const cats=new Set();
        for(let r=0;r<N;r++) cats.add(r*N+sols[0][r]);
        return {cats, regionOf, allSols};
      }
      const A=sols[0], B=sols[1];
      const S_B=[]; for(let r=0;r<N;r++) S_B.push(r*N+B[r]);
      const diffCells=[];
      for(let r=0;r<N;r++) if(A[r]!==B[r]) diffCells.push(r*N+B[r]);
      shuffle(diffCells);
      const killMoves=[], normMoves=[];
      for(const X of diffCells){
        const R=regionOf[X];
        if(!connectedAfterRemoval(N,regionOf,R,X)) continue;
        const nbs=neighbors4(X,N).filter(nb=>regionOf[nb]!==R);
        if(!nbs.length) continue;
        let kill=false;
        for(const nb of nbs){
          const rr=regionOf[nb];
          if(S_B.some(s=>s!==X && regionOf[s]===rr)){ kill=true; break; }
        }
        (kill?killMoves:normMoves).push(X);
      }
      const cand=killMoves.length?killMoves:normMoves;
      if(!cand.length) break;
      const X=cand[0];
      const R=regionOf[X];
      const nbs=neighbors4(X,N).filter(nb=>regionOf[nb]!==R);
      regionOf[X]=regionOf[nbs[Math.floor(Math.random()*nbs.length)]];
      const ns=solve2(N,regionOf,nl);
      if(ns===null){ regionOf[X]=R; break; } /* 太慢：棄盤 */
      if(ns.length===0){
        regionOf[X]=R; zeroStreak++;
        if(zeroStreak>25) break;
        continue;
      }
      zeroStreak=0; sols=ns;
    }
  }
  return null;
}

function transformBoard(N, regionOf){ /* 隨機旋轉/鏡像 ×8 */
  const T=Math.floor(Math.random()*8), ro=new Array(N*N);
  for(let r=0;r<N;r++)for(let c=0;c<N;c++){
    let r2=r,c2=c;
    if(T&1){ const t=r2; r2=c2; c2=t; }
    if(T&2){ r2=N-1-r2; }
    if(T&4){ c2=N-1-c2; }
    ro[r2*N+c2]=regionOf[r*N+c];
  }
  return ro;
}
function poolTake(n){
  if(!livePool[n] || !livePool[n].length) return null;
  const idx=Math.floor(Math.random()*livePool[n].length);
  const ro=livePool[n].splice(idx,1)[0];
  return transformBoard(n, ro);
}
let poolBusy=false;
function poolRefill(){ /* 只在閒置時補貨，不搶遊戲時間 */
  if(poolBusy) return;
  if(G && !G.tutorial && G.state==='marking' && !document.querySelector('.overlay.show')) return;
  const need=POOL_SIZES.filter(n=>(livePool[n]||[]).length<6);
  if(!need.length) return;
  poolBusy=true;
  setTimeout(()=>{
    const n=need[Math.floor(Math.random()*need.length)];
    const pz=generatePuzzle(n, n>=13?3000:1400);
    if(pz) livePool[n].push(Array.from(pz.regionOf));
    poolBusy=false;
  }, 30);
}

function fmtDur(ms){
  const s=Math.max(0,Math.ceil(ms/1000));
  const d=Math.floor(s/86400), h=Math.floor(s%86400/3600), m=Math.floor(s%3600/60), ss=s%60;
  if(d>=1) return d+' 天 '+h+' 小時';
  if(h>=1) return h+' 小時 '+m+' 分';
  return m+' 分 '+String(ss).padStart(2,'0')+' 秒';
}
function evalLock(){
  if(lockIv){ clearInterval(lockIv); lockIv=null; }

  if(s.banUntil && now<s.banUntil){ lockBan(s); return; }
  if(s.cdUntil && now<s.cdUntil){ lockCooldown(s); return; }
  if(lockNeeded()){ lockInput(); return; }
  hide('ov-lock');
  if(!G || G.state!=='marking'){ show('ov-menu'); refreshBest(); }
}

function newBoard(){
  const N = G.N;
  if(N>=10){
    const ro = poolTake(N);
    if(ro){
      const sol = N>=16 ? solve2(N, ro, 500000, 5) : solve2(N, ro);
      if(sol && sol.length>=1 && (N>=16 ? sol.length<=4 : sol.length===1)){
        const cats=new Set(); for(let r=0;r<N;r++) cats.add(r*N+sol[0][r]);
        finishBoard({cats, regionOf:ro, allSols: N>=16 ? sol : null});
        poolRefill();
        return;
      }
    }
  }
  $('gen-what').textContent = N+' × '+N+(G.tutorial?' 教學盤':' 唯一解盤面');
  show('ov-loading');
  setTimeout(()=>{
    let pz=null;
    if(N>=14){ pz=generatePuzzle(N,8000); }
    else { for(const b of [1200,3000,8000]){ pz=generatePuzzle(N,b); if(pz) break; } }
    hide('ov-loading');
    if(pz){ finishBoard(pz); poolRefill(); }
    else {
      G.state='done';
      const sub=$('genfail-sub');
      if(sub){ sub.textContent = N>=14 ? '大盤面的唯一解需要點運氣，建議重試一次，或先玩較小的盤面～' : '這次運氣不好，再試一次？'; }
      show('ov-genfail');
    }
  }, 60);
}

function finishBoard(pz){
  const N = G.N;
  G.cats = pz.cats; G.regionOf = pz.regionOf;
  /* 多解盤：全部解的星格並集（點中任一解的星都算貓） */
  G.catUnion = new Set();
  if(pz.allSols && pz.allSols.length){
    pz.allSols.forEach(s=>{ for(let r=0;r<N;r++) G.catUnion.add(r*N+s[r]); });
  } else { pz.cats.forEach(i=>G.catUnion.add(i)); }
  G.cells = [];
  for(let i=0;i<N*N;i++) G.cells.push({ cat: pz.cats.has(i), region: pz.regionOf[i], mark:'' });
  const pool = shuffle([...POKEMON]).slice(0, Math.max(N, G.catUnion.size));
  [...G.catUnion].sort((a,b)=>a-b).forEach((ci,k)=>{ G.cells[ci].poke = pool[k]; });

function neighbors8(i,N){
  const r=Math.floor(i/N), c=i%N, out=[];
  for(let dr=-1;dr<=1;dr++)for(let dc=-1;dc<=1;dc++){
    if(!dr&&!dc) continue;
    const rr=r+dr, cc=c+dc;
    if(rr>=0&&rr<N&&cc>=0&&cc<N) out.push(rr*N+cc);
  }
  return out;
}

function timerTick(){
  if(!TIMER.on) return;
  const now=performance.now();
  if(!timerPaused()) TIMER.left -= (now-TIMER.last)/1000;
  TIMER.last=now;
  if(TIMER.left<=0){ TIMER.left=0; renderTimer(); timeUp(); return; }

let GAME=null;

function wpick(arr,wf){
  const w=arr.map(wf); let s=w.reduce((a,b)=>a+b,0), r=Math.random()*s;
  for(let k=0;k<arr.length;k++){ r-=w[k]; if(r<=0) return arr[k]; }
  return arr[arr.length-1];
}
function prizeMonPool(){ return allMons().filter(p=>rarityOf(p.id)>=1 && !dexGot.has(p.id)); }
function rollMon(){
  /* v1.2：星數越多機率越低；棋盤格數越多、連勝越多 → 高星機率增加 */
  const N = G ? G.N : 6;
  const lvlBonus = Math.max(0, N-9);
  const streak = Math.min(20, ECON.streak||0);
  const W=[6, 3, 1.5, 0.7];
  return wpick(prizeMonPool(), p=>{
    const r=rarityOf(p.id);
    let w=W[r-1]||1;
    const lk=[0,0.18,0.30,0.45][r-1]||0;  /* 加成斜率：星越高吃越多 */
    const sk=[0,0.08,0.14,0.22][r-1]||0;
    w*= 1+lk*lvlBonus;
    w*= 1+sk*streak;
    return w;
  });
}
/* 抽獎核心：16% @@mon@@（稀有↑），其餘種子 1~5；luck1=星塵+1、luck2=必出@@mon@@（各消耗一次） */
function rollPrize(opt){
  opt=opt||{};
  let force=!!opt.forceMon;
  if(GAME && GAME.luck2){ force=true; GAME.luck2=false; }
  if(force || Math.random()<(opt.monC!=null?opt.monC:0.16)){
    const m=rollMon();
    if(m) return {type:'mon', mon:m};
  }

}

  const ctx = cv.getContext ? cv.getContext('2d') : null;
  if(!ctx){ setTimeout(reveal,500); return; }
  const gr=ctx.createLinearGradient(0,0,260,150);
  gr.addColorStop(0,'#cfd6de'); gr.addColorStop(.5,'#aab4bf'); gr.addColorStop(1,'#8d99a6');
  ctx.fillStyle=gr; ctx.fillRect(0,0,260,150);

  ctx.fillText('🪙 刮這裡 🪙',130,82);
  let down=false, moves=0;
  const pt=e=>{ const r=cv.getBoundingClientRect(); return {x:(e.clientX-r.left)*(260/Math.max(1,r.width)), y:(e.clientY-r.top)*(150/Math.max(1,r.height))}; };
  const scratchAt=e=>{
    const p=pt(e);
    ctx.globalCompositeOperation='destination-out';
    ctx.beginPath(); ctx.arc(p.x,p.y,17,0,7); ctx.fill();
    if(++moves%6===0){
      const d=ctx.getImageData(0,0,260,150).data;

function waitCells(cells){ /* 等玩家把指定格全部 ✕（任一順序），走正常點擊流程 */
  return new Promise(res=>{

  });
}

function demoAlive(){ return G && G.tutorial && G.state!=='done'; }

function findForced(){ /* 找「真的可推理」的下一隻 */
  const N=G.N;
  const caught=[]; G.cells.forEach((c,i)=>{ if(c.revealed) caught.push(i); });
  const banned=new Set();
  caught.forEach(ci=>neighbors8(ci,N).forEach(nb=>banned.add(nb)));
  const isCand=i=>!G.cells[i].mark && !G.cells[i].revealed && !banned.has(i);
  const stars=[]; G.cells.forEach((c,i)=>{ if(c.cat && !c.revealed) stars.push(i); });
  for(const s of stars){
    const r=Math.floor(s/N), c=s%N, rg=G.cells[s].region;
    let rowC=0,colC=0,regC=0, rowOk=false,colOk=false,regOk=false;
    for(let i=r*N;i<r*N+N;i++) if(isCand(i)){ rowC++; if(i===s) rowOk=true; }
    for(let i=c;i<N*N;i+=N) if(isCand(i)){ colC++; if(i===s) colOk=true; }
    for(let i=0;i<N*N;i++) if(G.cells[i].region===rg && isCand(i)){ regC++; if(i===s) regOk=true; }
    if(rowC===1&&rowOk) return {cell:s, why:'第'+(r+1)+'列'};
    if(colC===1&&colOk) return {cell:s, why:'第'+(c+1)+'欄'};
    if(regC===1&&regOk) return {cell:s, why:'這個顏色區塊'};
  }
  return null;
}

async function demoEliminate(star, gap){
  const N=G.N, r=Math.floor(star/N), c=star%N;
  tutShow('接著推理：牠在第 <b>'+(r+1)+' 列</b> → 這列其他格不可能有 → ✕');
  for(let i=r*N;i<r*N+N;i++){ if(i!==star){ demoMark(i); await sleep(gap); } }
  if(!demoAlive()) return;
  tutShow('第 <b>'+(c+1)+' 欄</b>也一樣 ✕');
  for(let i=c;i<N*N;i+=N){ if(i!==star){ demoMark(i); await sleep(gap); } }
  if(!demoAlive()) return;
  const rg=G.cells[star].region;
  const regCells=[]; G.cells.forEach((cc,idx)=>{ if(cc.region===rg && idx!==star && cc.mark!=='x') regCells.push(idx); });
  if(regCells.length){
    tutShow('牠的<b>顏色區塊</b>裡也不會有第二隻 ✕');
    for(const i of regCells){ demoMark(i); await sleep(gap); }
  }
}
